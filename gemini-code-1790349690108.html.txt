    <script>
    class SoundManager {
        constructor() {
            this.ctx = null;
            this.enabled = true;
        }

        init() {
            if (!this.ctx) {
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                if (AudioCtx) this.ctx = new AudioCtx();
            }
            if (this.ctx && this.ctx.state === 'suspended') {
                this.ctx.resume();
            }
        }

        playShoot(type = 'normal') {
            if (!this.enabled || !this.ctx) return;
            const t = this.ctx.currentTime;
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();

    class AccountManager {
        constructor() {
            this.storageKey = 'trolley_tankers_accounts_v4';
            this.activeKey = 'trolley_tankers_active_user_v4';
            this.accounts = this.loadAll();
            this.activeAccount = this.loadActive();
        }

        loadAll() {
            try {
                const data = localStorage.getItem(this.storageKey);
                if (data) return JSON.parse(data);
            } catch (e) {}
            return {
                'Coupon Commando': {
                    name: 'Coupon Commando',
                    createdAt: Date.now(),
                    coupons: 60,
                    kills: 0,
                    pushersKilled: 0,
                    wins: 0,
                    matchesPlayed: 0,
                    damageDealt: 0,
                    powerupsCollected: 0,
                    selectedSkin: 'standard',
                    unlockedSkins: ['standard']
                }
            };
        }

        saveAll() {
            try {
                localStorage.setItem(this.storageKey, JSON.stringify(this.accounts));
                if (this.activeAccount) {
                    localStorage.setItem(this.activeKey, this.activeAccount.name);
                }
            } catch (e) {}
        }

        loadActive() {
            const savedName = localStorage.getItem(this.activeKey);
            if (savedName && this.accounts[savedName]) {
                return this.accounts[savedName];
            }
            const keys = Object.keys(this.accounts);
            return keys.length > 0 ? this.accounts[keys[0]] : null;
        }

        setActive(name) {
            if (this.accounts[name]) {
                this.activeAccount = this.accounts[name];
                this.saveAll();
                this.updateUI();
                return true;
            }
            return false;
        }

        createAccount(name) {
            const cleanName = name.trim();
            if (!cleanName || this.accounts[cleanName]) return false;
            const newAcc = {
                name: cleanName,
                createdAt: Date.now(),
                coupons: 35,
                kills: 0,
                pushersKilled: 0,
                wins: 0,
                matchesPlayed: 0,
                damageDealt: 0,
                powerupsCollected: 0,
                selectedSkin: 'standard',
                unlockedSkins: ['standard']
            };
            this.accounts[cleanName] = newAcc;
            this.activeAccount = newAcc;
            this.saveAll();
            this.updateUI();
            return true;
        }

        getTier(acc) {
            const total = (acc.wins * 3) + acc.kills + (acc.pushersKilled * 0.5);
            if (total >= 40) return { name: 'DIAMOND TYCOON', color: 'text-cyan-700 bg-cyan-100 border-cyan-300' };
            if (total >= 20) return { name: 'GOLD BULK BUYER', color: 'text-amber-800 bg-amber-100 border-amber-300' };
            if (total >= 8) return { name: 'SILVER SAVER', color: 'text-slate-800 bg-slate-200 border-slate-300' };
            return { name: 'BRONZE SHOPPER', color: 'text-amber-900 bg-orange-100 border-orange-300' };
        }

        unlockSkin(skinId) {
            const skin = TROLLEY_SKINS.find(s => s.id === skinId);
            if (!skin || !this.activeAccount) return false;
            if (this.activeAccount.unlockedSkins.includes(skinId)) {
                this.activeAccount.selectedSkin = skinId;
                this.saveAll();
                this.updateUI();
                return true;
            }
            if (this.activeAccount.coupons >= skin.price) {
                this.activeAccount.coupons -= skin.price;
                this.activeAccount.unlockedSkins.push(skinId);
                this.activeAccount.selectedSkin = skinId;
                this.saveAll();
                this.updateUI();
                sound.playRegisterChime();
                return true;
            }
            return false;
        }

        recordMatch(kills, pushersKilled, damage, powerups, won) {
            if (!this.activeAccount) return 0;
            this.activeAccount.matchesPlayed++;
            this.activeAccount.kills += kills;
            this.activeAccount.pushersKilled = (this.activeAccount.pushersKilled || 0) + pushersKilled;
            this.activeAccount.damageDealt += Math.round(damage);
            this.activeAccount.powerupsCollected += powerups;
            if (won) this.activeAccount.wins++;

            const earnedCoupons = (kills * 12) + (pushersKilled * 6) + (won ? 30 : 8) + (powerups * 4);
            this.activeAccount.coupons += earnedCoupons;
            this.saveAll();
            this.updateUI();
            return earnedCoupons;
        }

    function initMultiplayer() {
        const randomCode = 'cart-' + Math.floor(1000 + Math.random() * 9000);
        peer = new Peer(randomCode);

        peer.on('open', (id) => {
            const myPeerBox = document.getElementById('my-peer-id');
            if (myPeerBox) myPeerBox.value = id;
            const roomBadge = document.getElementById('header-room-code');
            if (roomBadge) roomBadge.innerText = id;
        });

        peer.on('connection', (conn) => {
            setupConnectionHandlers(conn);
            postKillfeed(`🌐 Another shopper entered your aisle!`, 'text-blue-600 font-bold');
        });

        peer.on('error', (err) => {
            console.error('PeerJS error:', err);
        });
    }

    function connectToFriend(targetCode) {
        if (!peer || !targetCode) return;
        const conn = peer.connect(targetCode.trim());
        setupConnectionHandlers(conn);
        conn.on('open', () => {
            postKillfeed(`🌐 Connected to Aisle: ${targetCode}!`, 'text-blue-600 font-bold');
            const roomBadge = document.getElementById('header-room-code');
            if (roomBadge) roomBadge.innerText = targetCode;
        });
    }

    function setupConnectionHandlers(conn) {
        conn.on('open', () => {
            connections.push(conn);
            isMultiplayerActive = true;
            document.getElementById('multiplayer-modal').classList.add('hidden');
            if (!isRunning) startBattle();
        });

        conn.on('data', (data) => {
            handleNetworkMessage(data);
        });

        conn.on('close', () => {
            connections = connections.filter(c => c !== conn);
            if (connections.length === 0) {
                isMultiplayerActive = false;
            }
        });
    }

    function broadcastNetMessage(msg) {
        for (let conn of connections) {
            if (conn.open) conn.send(msg);
        }
    }

    function broadcastLocalState() {
        const p1 = players.find(p => p.id === 'p1');

    function handleNetworkMessage(data) {
        if (data.type === 'player_sync') {
            let remotePlayer = players.find(p => p.id === data.id);
            if (!remotePlayer) {
                remotePlayer = new TrolleyTank(data.id, data.x, data.y, data.color, false, data.name, data.skinId);
                players.push(remotePlayer);
                postKillfeed(`🛒 ${data.name} rolled into the battle!`, 'text-emerald-600 font-bold');
            }
            remotePlayer.x = data.x;
            remotePlayer.y = data.y;
            remotePlayer.vx = data.vx;
            remotePlayer.vy = data.vy;
            remotePlayer.heading = data.heading;
            remotePlayer.turretAngle = data.turretAngle;
            remotePlayer.cartHealth = data.cartHealth;
            remotePlayer.pusherHealth = data.pusherHealth;
            remotePlayer.isPusherDead = data.isPusherDead;
            remotePlayer.isDead = data.isDead;
            remotePlayer.weaponType = data.weaponType;
            remotePlayer.ammo = data.ammo;
        } else if (data.type === 'remote_fire') {
            const wpn = WEAPONS[data.weaponType] || WEAPONS.normal;
            bullets.push(new Bullet(data.id, data.muzzleX, data.muzzleY, data.angle, wpn, data.color, wpn.damage));
            sound.playShoot(data.weaponType);
        }
    }

    function getCurrentZone(x, y) {
        for (let zone of SUPERSTORE_ZONES) {
            const b = zone.bounds;
            if (x >= b.x && x <= b.x + b.w && y >= b.y && y <= b.y + b.h) {
                return zone;
            }
        }
        return SUPERSTORE_ZONES[0];
    }

    class TrolleyTank {
        constructor(id, x, y, color, isBot = false, name = 'Shopper', skinId = 'standard') {
            this.id = id;
            this.x = x;
            this.y = y;
            this.vx = 0;
            this.vy = 0;
            this.heading = Math.random() * Math.PI * 2;
            this.turretAngle = this.heading;
            this.radius = 24;
            this.skinId = skinId;
            this.color = color;
            this.isBot = isBot;
            this.name = name;

            this.maxCartHealth = 100;
            this.cartHealth = 100;

            this.maxPusherHealth = 85;
            this.pusherHealth = 85;
            this.isPusherDead = false;

            this.shield = 0;
            this.isDead = false;

            this.speed = 0;
            this.accel = 0.36;
            this.friction = 0.93;
            this.steerSpeed = 0.06;

            this.nitro = 100;
            this.isBoosting = false;

            this.weaponType = 'normal';
            this.ammo = Infinity;
            this.fireCooldown = 0;
            this.score = 0;

            this.casterAngle = 0;
            this.squeakTimer = Math.random() * 50;

            this.aiTarget = null;
            this.aiRetargetTimer = 0;
            this.aiShootTimer = 0;

            this.pusherDist = 42; 
            this.pusherRadius = 23;
            this.pusherWalkCycle = 0;
        }

        getPusherPos() {
            return {
                x: this.x - Math.cos(this.heading) * this.pusherDist,
                y: this.y - Math.sin(this.heading) * this.pusherDist
            };
        }

        update(dt) {
            if (this.isDead) return;

            const speedSq = this.vx * this.vx + this.vy * this.vy;
            if (speedSq > 3.0 && !this.isPusherDead) {
                this.squeakTimer--;
                if (this.squeakTimer <= 0) {
                    sound.playSqueak();
                    this.squeakTimer = 40 + Math.random() * 50;
                }
                this.pusherWalkCycle += Math.sqrt(speedSq) * 0.2;
            }

            if (this.id === 'p1' && !this.isDead) {
                const wpn = WEAPONS[this.weaponType];
                if (wpn.isRapidFire && (isMouseDown || isTouchFiring || keys['Space'])) {
                    this.fire();
                }
            }

            if (!this.isPusherDead) {
                if (this.isBoosting && this.nitro > 0) {
                    this.nitro = Math.max(0, this.nitro - 1.3);
                    this.vx += Math.cos(this.heading) * 0.58;
                    this.vy += Math.sin(this.heading) * 0.58;

                    if (Math.random() < 0.6) {
                        particles.push(new Particle(
                            this.x - Math.cos(this.heading) * 44 + (Math.random() - 0.5) * 10,
                            this.y - Math.sin(this.heading) * 44 + (Math.random() - 0.5) * 10,
                            -Math.cos(this.heading) * 3.5 + (Math.random() - 0.5) * 2,
                            -Math.sin(this.heading) * 3.5 + (Math.random() - 0.5) * 2,
                            '#0284c7',
                            Math.random() * 4 + 2,
                            18
                        ));
                    }
                } else {
                    this.nitro = Math.min(100, this.nitro + 0.35);
                }
            }

            const currentZone = getCurrentZone(this.x, this.y);
            let curFriction = currentZone.slippery ? 0.982 : this.friction;

            if (this.isPusherDead) {
                curFriction = 0.88;
            }

            for (let belt of conveyorBelts) {
                if (this.x >= belt.x && this.x <= belt.x + belt.w &&
                    this.y >= belt.y && this.y <= belt.y + belt.h) {
                    this.vx += belt.vx;
                    this.vy += belt.vy;
                }
            }

            const pPos = this.getPusherPos();
            for (let puddle of firePuddles) {
                const distCart = Math.hypot(this.x - puddle.x, this.y - puddle.y);
                const distPusher = Math.hypot(pPos.x - puddle.x, pPos.y - puddle.y);

                if (distCart < puddle.radius) {
                    this.takeDamage(0.35, 'cart', puddle.ownerId);
                }
                if (distPusher < puddle.radius + 10 && !this.isPusherDead) {
                    this.takeDamage(0.65, 'pusher', puddle.ownerId);
                }
            }

            this.x += this.vx;
            this.y += this.vy;
            this.vx *= curFriction;
            this.vy *= curFriction;

            if (this.fireCooldown > 0) {
                this.fireCooldown--;
            }

            const pad = this.radius + 24;
            if (this.x < pad) { this.x = pad; this.vx = Math.abs(this.vx) * 0.5; }
            if (this.x > WORLD_WIDTH - pad) { this.x = WORLD_WIDTH - pad; this.vx = -Math.abs(this.vx) * 0.5; }
            if (this.y < pad) { this.y = pad; this.vy = Math.abs(this.vy) * 0.5; }
            if (this.y > WORLD_HEIGHT - pad) { this.y = WORLD_HEIGHT - pad; this.vy = -Math.abs(this.vy) * 0.5; }

            this.resolveObstacleCollisions();

            if (speedSq > 16 && Math.random() < 0.28) {
                tireTracks.push({
                    x: this.x - Math.cos(this.heading) * 14,
                    y: this.y - Math.sin(this.heading) * 14,
                    angle: this.heading,
                    alpha: 0.3,
                    life: 240
                });
            }
        }

        resolveObstacleCollisions() {
            for (let obs of obstacles) {
                const closestX = Math.max(obs.x, Math.min(this.x, obs.x + obs.w));
                const closestY = Math.max(obs.y, Math.min(this.y, obs.y + obs.h));
                const dx = this.x - closestX;
                const dy = this.y - closestY;
                const distSq = dx * dx + dy * dy;

                if (distSq < this.radius * this.radius) {
                    const dist = Math.sqrt(distSq) || 0.001;
                    const overlap = this.radius - dist;
                    const nx = dx / dist;
                    const ny = dy / dist;

                    this.x += nx * overlap;
                    this.y += ny * overlap;

                    const dot = this.vx * nx + this.vy * ny;
                    if (dot < 0) {
                        this.vx = (this.vx - 1.5 * dot * nx) * 0.65;
                        this.vy = (this.vy - 1.5 * dot * ny) * 0.65;
                    }
                }
            }

            for (let c of circularObstacles) {
                const dx = this.x - c.x;
                const dy = this.y - c.y;
                const dist = Math.hypot(dx, dy) || 0.001;
                const minDist = this.radius + c.radius;
                if (dist < minDist) {
                    const overlap = minDist - dist;
                    const nx = dx / dist;
                    const ny = dy / dist;
                    this.x += nx * overlap;
                    this.y += ny * overlap;

                    const dot = this.vx * nx + this.vy * ny;
                    if (dot < 0) {
                        this.vx = (this.vx - 1.4 * dot * nx) * 0.65;
                        this.vy = (this.vy - 1.4 * dot * ny) * 0.65;
                    }
                }
            }
        }

        drive(forward = true) {
            if (this.isPusherDead || this.isDead) return;

            const dir = forward ? 1 : -0.55;
            const force = this.accel * dir;
            this.vx += Math.cos(this.heading) * force;
            this.vy += Math.sin(this.heading) * force;
            this.casterAngle += 0.3;
        }

        turn(direction) {
            if (this.isPusherDead || this.isDead) return;
            this.heading += this.steerSpeed * direction;
        }

        setAim(angle) {
            this.turretAngle = angle;
        }

        fire() {
            if (this.fireCooldown > 0 || this.isDead) return false;

            const wpn = WEAPONS[this.weaponType];
            this.fireCooldown = wpn.cooldown;

            const recoilAngle = this.turretAngle + Math.PI;
            this.vx += Math.cos(recoilAngle) * wpn.recoil;
            this.vy += Math.sin(recoilAngle) * wpn.recoil;

            const barrelLen = 34;
            const muzzleX = this.x + Math.cos(this.turretAngle) * barrelLen;
            const muzzleY = this.y + Math.sin(this.turretAngle) * barrelLen;

            let finalAngle = this.turretAngle;
            if (wpn.spread) {
                finalAngle += (Math.random() - 0.5) * wpn.spread;
            }

            if (wpn.shotgunPellets) {
                const spread = 0.35;
                for (let k = 0; k < wpn.shotgunPellets; k++) {
                    const pelletAngle = this.turretAngle - spread / 2 + (spread / (wpn.shotgunPellets - 1)) * k + (Math.random() - 0.5) * 0.08;
                    bullets.push(new Bullet(this.id, muzzleX, muzzleY, pelletAngle, wpn, this.color, wpn.damage));
                }
            } else {
                bullets.push(new Bullet(this.id, muzzleX, muzzleY, finalAngle, wpn, this.color, wpn.damage));
            }

            if (this.id === 'p1' && connections.length > 0) {
                broadcastNetMessage({
                    type: 'remote_fire',
                    id: peer ? peer.id : 'p1',
                    muzzleX: muzzleX,
                    muzzleY: muzzleY,
                    angle: finalAngle,
                    weaponType: this.weaponType,
                    color: this.color
                });
            }

            const ejectAngle = this.turretAngle + Math.PI * 0.5;
            shellCasings.push(new ShellCasing(
                this.x, this.y,
                Math.cos(ejectAngle) * 3, Math.sin(ejectAngle) * 3
            ));

            const pCount = wpn.isRapidFire ? 2 : 6;
            for (let i = 0; i < pCount; i++) {
                particles.push(new Particle(
                    muzzleX, muzzleY,
                    Math.cos(this.turretAngle + (Math.random() - 0.5) * 0.6) * (3 + Math.random() * 5),
                    Math.sin(this.turretAngle + (Math.random() - 0.5) * 0.6) * (3 + Math.random() * 5),
                    wpn.color, Math.random() * 2 + 1.5, 12
                ));
            }

            sound.playShoot(this.weaponType);

            if (this.weaponType !== 'normal') {
                this.ammo--;
                if (this.ammo <= 0) {
                    this.weaponType = 'normal';
                    this.ammo = Infinity;
                    postKillfeed(`${this.name}'s special weapon ran dry!`, 'text-slate-500');
                }
            }

            return true;
        }

        takeDamage(amount, targetPart = 'cart', attackerId = null) {
            if (this.isDead) return;

            if (this.shield > 0) {
                if (this.shield >= amount) {
                    this.shield -= amount;
                    amount = 0;
                } else {
                    amount -= this.shield;
                    this.shield = 0;
                    postKillfeed(`${this.name}'s Kevlar Bag popped!`, 'text-purple-600');
                }
            }

            if (amount <= 0) return;

            if (targetPart === 'pusher' && !this.isPusherDead) {
                this.pusherHealth -= amount;
                for (let i = 0; i < 4; i++) {
                    const pPos = this.getPusherPos();
                    particles.push(new Particle(pPos.x, pPos.y, (Math.random() - 0.5) * 5, (Math.random() - 0.5) * 5, '#ef4444', 3.5, 22));
                }

                if (this.pusherHealth <= 0) {
                    this.pusherHealth = 0;
                    this.isPusherDead = true;
                    sound.playPusherScream();

                    const killer = players.find(p => p.id === attackerId);
                    const killerName = killer ? killer.name : 'Aisle Crossfire';
                    postKillfeed(`💀 ${killerName} DROPPED ${this.name}'s BIG PUSHER! (IMMOBILIZED!)`, 'text-rose-600 font-bold');

                    if (killer && killer.id === 'p1') {
                        stats.pushersKilled++;
                    }

                    if (this.id === 'p1') {
                        const warn = document.getElementById('pusher-status-warning');
                        if (warn) warn.classList.remove('hidden');
                    }
                }
            } else {
                this.cartHealth -= amount;
                for (let i = 0; i < 4; i++) {
                    particles.push(new Particle(this.x, this.y, (Math.random() - 0.5) * 5, (Math.random() - 0.5) * 5, '#64748b', 3, 18));
                }

                if (this.cartHealth <= 0) {
                    this.cartHealth = 0;
                    this.die(attackerId);
                }
            }
        }

        die(attackerId) {
            this.isDead = true;
            sound.playExplosion(true);
            sound.playRegisterChime();

            for (let i = 0; i < 40; i++) {
                const colors = ['#ef4444', '#f59e0b', '#64748b', '#cbd5e1', '#ffffff', '#eab308'];
                particles.push(new Particle(
                    this.x, this.y,
                    (Math.random() - 0.5) * 14, (Math.random() - 0.5) * 14,
                    colors[Math.floor(Math.random() * colors.length)],
                    Math.random() * 6 + 3, 48
                ));
            }

            const killer = players.find(p => p.id === attackerId);
            const killerName = killer ? killer.name : 'Supermarket Hazard';
            postKillfeed(`💥 ${killerName} obliterated ${this.name}'s Cart!`, 'text-rose-600 font-black');

            if (killer && killer.id !== this.id) {
                killer.score += 45.00;
                if (killer.id === 'p1') {
                    stats.kills++;
                }
            }

            checkRoundStatus();
        }

    class Bullet {
        constructor(ownerId, x, y, angle, config, trailColor, customDmg = null) {
            this.ownerId = ownerId;
            this.x = x;
            this.y = y;
            this.angle = angle;
            this.config = config;
            this.vx = Math.cos(angle) * config.speed;
            this.vy = Math.sin(angle) * config.speed;
            this.radius = config.radius;
            this.damage = customDmg !== null ? customDmg : config.damage;
            this.splashRadius = config.splashRadius;
            this.trailColor = trailColor;
            this.life = config.isRapidFire ? 120 : (config.isBallisticMortar ? 220 : 200);
            this.isAlive = true;
            this.penetration = config.penetration || 1;
            this.ricochets = config.ricochets || 0;
            this.isBallisticMortar = config.isBallisticMortar || false;
            this.shrapnelCount = config.shrapnelCount || 0;
            this.leavesFirePuddle = config.leavesFirePuddle || false;

            this.maxLife = this.life;
            this.arcHeight = 0;
        }

        update() {
            this.x += this.vx;
            this.y += this.vy;
            this.life--;

            if (this.isBallisticMortar) {
                const progress = 1 - (this.life / this.maxLife);
                this.arcHeight = Math.sin(progress * Math.PI) * 32;

                if (Math.random() < 0.35) {
                    particles.push(new Particle(this.x, this.y - this.arcHeight, (Math.random() - 0.5) * 1.5, (Math.random() - 0.5) * 1.5, '#16a34a', 2.5, 14));
                }

                if (this.life <= 145) {
                    this.explode(null);
                    return;
                }
            }

            if (this.life <= 0) {
                this.explode(null);
                return;
            }

            if (this.x < this.radius || this.x > WORLD_WIDTH - this.radius ||
                this.y < this.radius || this.y > WORLD_HEIGHT - this.radius) {
                if (this.ricochets > 0) {
                    this.ricochets--;
                    if (this.x < this.radius || this.x > WORLD_WIDTH - this.radius) this.vx *= -1;
                    if (this.y < this.radius || this.y > WORLD_HEIGHT - this.radius) this.vy *= -1;
                    this.angle = Math.atan2(this.vy, this.vx);
                    sound.playHit();
                } else {
                    this.explode(null);
                    return;
                }
            }

            if (!this.isBallisticMortar) {
                for (let obs of obstacles) {
                    if (this.x >= obs.x && this.x <= obs.x + obs.w &&
                        this.y >= obs.y && this.y <= obs.y + obs.h) {
                        if (this.ricochets > 0) {
                            this.ricochets--;
                            this.vx *= -1;
                            this.vy *= -1;
                            this.angle = Math.atan2(this.vy, this.vx);
                            sound.playHit();
                        } else {
                            this.explode(null);
                            return;
                        }
                        break;
                    }
                }
            }

            for (let target of players) {
                if (target.id === this.ownerId || target.isDead) continue;

                if (!target.isPusherDead) {
                    const pPos = target.getPusherPos();
                    const distToPusher = Math.hypot(this.x - pPos.x, this.y - pPos.y);
                    if (distToPusher < target.pusherRadius + this.radius) {
                        target.takeDamage(this.damage * 1.25, 'pusher', this.ownerId);
                        if (this.ownerId === 'p1') stats.damageDealt += this.damage * 1.25;
                        this.explode(target);
                        return;
                    }
                }

                const distToCart = Math.hypot(this.x - target.x, this.y - target.y);
                if (distToCart < target.radius + this.radius) {
                    target.takeDamage(this.damage, 'cart', this.ownerId);
                    if (this.ownerId === 'p1') stats.damageDealt += this.damage;

                    if (this.config.knockback) {
                        target.vx += Math.cos(this.angle) * this.config.knockback;
                        target.vy += Math.sin(this.angle) * this.config.knockback;
                    }

                    this.penetration--;
                    if (this.penetration <= 0) {
                        this.explode(target);
                        return;
                    }
                }
            }
        }

        explode(directTarget) {
            this.isAlive = false;

            if (this.splashRadius > 20) {
                sound.playExplosion(this.splashRadius > 80);
                for (let p of players) {
                    if (p.isDead || p === directTarget) continue;

                    const distCart = Math.hypot(p.x - this.x, p.y - this.y);
                    if (distCart < this.splashRadius) {
                        const falloff = 1 - (distCart / this.splashRadius);
                        const aoe = Math.round(this.damage * falloff * 0.7);
                        p.takeDamage(aoe, 'cart', this.ownerId);
                        if (this.ownerId === 'p1') stats.damageDealt += aoe;
                    }

                    if (!p.isPusherDead) {
                        const pPos = p.getPusherPos();
                        const distPusher = Math.hypot(pPos.x - this.x, pPos.y - this.y);
                        if (distPusher < this.splashRadius) {
                            const falloff = 1 - (distPusher / this.splashRadius);
                            const aoe = Math.round(this.damage * falloff * 0.8);

    class ProximityMine {
        constructor(ownerId, x, y) {
            this.ownerId = ownerId;
            this.x = x;
            this.y = y;
            this.triggerRadius = 55;
            this.life = 1800;
            this.isArming = 40;
        }

        update() {
            if (this.isArming > 0) {
                this.isArming--;
                return;
            }
            this.life--;

            for (let p of players) {
                if (p.isDead) continue;
                if (Math.hypot(p.x - this.x, p.y - this.y) < this.triggerRadius) {
                    this.detonate();
                    return;
                }
            }
        }

        detonate() {
            sound.playExplosion(true);
            for (let p of players) {
                if (p.isDead) continue;
                const d = Math.hypot(p.x - this.x, p.y - this.y);
                if (d < 110) {
                    const dmg = Math.round(55 * (1 - d / 110));
                    p.takeDamage(dmg, 'cart', this.ownerId);
                    p.takeDamage(dmg * 0.8, 'pusher', this.ownerId);
                    if (this.ownerId === 'p1') stats.damageDealt += dmg;
                }
            }
            this.life = 0;
        }

        draw(ctx) {
            ctx.save();
            ctx.translate(this.x, this.y);
            ctx.fillStyle = 'rgba(56, 189, 248, 0.2)';
            ctx.beginPath();
            ctx.arc(0, 0, this.triggerRadius, 0, Math.PI * 2);
            ctx.fill();
            ctx.font = '16px serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('🦃', 0, 0);
            ctx.restore();
        }
    }

    class Particle {
        constructor(x, y, vx, vy, color, size, maxLife) {
            this.x = x;
            this.y = y;
            this.vx = vx;
            this.vy = vy;
            this.color = color;
            this.size = size;
            this.maxLife = maxLife;
            this.life = maxLife;
        }

        update() {
            this.x += this.vx;
            this.y += this.vy;
            this.vx *= 0.94;
            this.vy *= 0.94;
            this.life--;
        }

        draw(ctx) {
            const alpha = Math.max(0, this.life / this.maxLife);
            ctx.save();
            ctx.globalAlpha = alpha;
            ctx.fillStyle = this.color;
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size * (this.life / this.maxLife), 0, Math.PI * 2);
            ctx.fill();
            ctx.restore();
        }
    }

    class ShellCasing {
        constructor(x, y, vx, vy) {
            this.x = x;
            this.y = y;
            this.vx = vx;
            this.vy = vy;
            this.angle = Math.random() * Math.PI * 2;
            this.life = 250;
        }

        update() {
            if (this.life > 210) {
                this.x += this.vx;
                this.y += this.vy;
                this.vx *= 0.9;
                this.vy *= 0.9;
            }
            this.life--;
        }

        draw(ctx) {
            if (this.life <= 0) return;
            ctx.save();
            ctx.translate(this.x, this.y);
            ctx.rotate(this.angle);
            ctx.globalAlpha = Math.min(1, this.life / 50);
            ctx.fillStyle = '#ca8a04';
            ctx.fillRect(-3, -1.2, 6, 2.4);
            ctx.restore();
        }
    }

            const shelfDepth = Math.max(16, width - 12);
            const itemSpacing = 16;

    function spawnPowerup() {
        if (powerups.length >= MAX_ACTIVE_POWERUPS) return;

        const proto = POWERUP_CATALOG[Math.floor(Math.random() * POWERUP_CATALOG.length)];

        let attempts = 0;
        let px = 0, py = 0;
        let valid = false;

        while (attempts < 30 && !valid) {
            attempts++;
            px = 180 + Math.random() * (WORLD_WIDTH - 360);
            py = 180 + Math.random() * (WORLD_HEIGHT - 360);

            let collides = false;
            for (let obs of obstacles) {
                if (px >= obs.x - 35 && px <= obs.x + obs.w + 35 &&
                    py >= obs.y - 35 && py <= obs.y + obs.h + 35) {
                    collides = true;
                    break;
                }
            }
            if (collides) continue;

            let tooClose = false;
            for (let p of powerups) {
                if (Math.hypot(p.x - px, p.y - py) < 450) {
                    tooClose = true;
                    break;
                }
            }
            if (!tooClose) valid = true;
        }

        if (!valid) return;

        powerups.push({
            x: px,
            y: py,
            radius: 22,
            type: proto.type,
            label: proto.label,
            icon: proto.icon,
            color: proto.color,
            bobble: Math.random() * Math.PI * 2
        });

        const z = getCurrentZone(px, py);
        showStoreWideBanner(`📦 ${proto.label.toUpperCase()} DROPPED IN ${z.name.toUpperCase()}!`);
        postKillfeed(`📦 ${proto.label} arrived in ${z.name}!`, 'text-amber-700 font-bold');
    }

    function drawFloor(ctx) {

            const startC = Math.floor(Math.max(b.x, camera.x) / tileSize);
            const endC = Math.ceil(Math.min(b.x + b.w, camera.x + camera.width) / tileSize);
            const startR = Math.floor(Math.max(b.y, camera.y) / tileSize);
            const endR = Math.ceil(Math.min(b.y + b.h, camera.y + camera.height) / tileSize);

            for (let c = startC; c < endC; c++) {
                for (let r = startR; r < endR; r++) {

                        ctx.fillRect(c * tileSize + 10, r * tileSize + 10, 24, 6);
                    }
                }

            ctx.strokeStyle = zone.trimColor;
            ctx.lineWidth = 3;
            ctx.strokeRect(b.x + 2, b.y + 2, b.w - 4, b.h - 4);

            ctx.save();
            ctx.font = '900 28px Bungee, sans-serif';

    function updateShopperBotAI(bot, dt) {
        if (bot.isDead) return;

        bot.aiRetargetTimer--;
        if (bot.aiRetargetTimer <= 0 || !bot.aiTarget || bot.aiTarget.isDead) {
            if (powerups.length > 0 && Math.random() < 0.5) {
                bot.aiTarget = powerups[0];
            } else {
                let closest = null;
                let minDist = Infinity;
                for (let p of players) {
                    if (p.id !== bot.id && !p.isDead) {
                        const d = Math.hypot(p.x - bot.x, p.y - bot.y);
                        if (d < minDist) {
                            minDist = d;
                            closest = p;
                        }
                    }
                }
                bot.aiTarget = closest;
            }
            bot.aiRetargetTimer = 45 + Math.random() * 30;
        }

        if (!bot.aiTarget) return;

        const target = bot.aiTarget;
        const dx = target.x - bot.x;
        const dy = target.y - bot.y;
        const dist = Math.hypot(dx, dy);
        let targetAngle = Math.atan2(dy, dx);

        const probeX = bot.x + Math.cos(bot.heading) * 75;
        const probeY = bot.y + Math.sin(bot.heading) * 75;
        let blocked = false;

        for (let obs of obstacles) {
            if (probeX >= obs.x && probeX <= obs.x + obs.w &&
                probeY >= obs.y && probeY <= obs.y + obs.h) {
                blocked = true;
                break;
            }
        }
        if (blocked) targetAngle += Math.PI / 2;

        let diff = targetAngle - bot.heading;
        while (diff < -Math.PI) diff += Math.PI * 2;
        while (diff > Math.PI) diff -= Math.PI * 2;

        if (!bot.isPusherDead) {
            if (diff > 0.12) bot.turn(1);
            else if (diff < -0.12) bot.turn(-1);
            if (Math.abs(diff) < 1.1) bot.drive(true);
            bot.isBoosting = dist > 550 && bot.nitro > 40 && Math.random() < 0.1;
        }

        if (target instanceof TrolleyTank) {
            let aimX = target.x;
            let aimY = target.y;

            if (!target.isPusherDead && botDifficulty !== 'easy' && Math.random() < 0.6) {
                const pPos = target.getPusherPos();
                aimX = pPos.x;
                aimY = pPos.y;
            }

            const lead = botDifficulty === 'hard' ? 7 : 3;
            aimX += target.vx * lead;
            aimY += target.vy * lead;

            bot.setAim(Math.atan2(aimY - bot.y, aimX - bot.x));

            bot.aiShootTimer--;
            if (bot.aiShootTimer <= 0 && dist < 600) {
                const aimDiff = Math.abs(bot.turretAngle - Math.atan2(target.y - bot.y, target.x - bot.x));
                if (aimDiff < 0.45) {
                    bot.fire();
                    const wpn = WEAPONS[bot.weaponType];
                    bot.aiShootTimer = wpn.isRapidFire ? 5 : (botDifficulty === 'hard' ? 24 : 40);
                }
            }
        }
    }

    function updateDragAimCalculations(clientX, clientY) {
        const rect = canvas.getBoundingClientRect();
        const cx = clientX - rect.left;
        const cy = clientY - rect.top;

        dragAim.currentX = cx;
        dragAim.currentY = cy;
        dragAim.worldCurrentX = cx + camera.x;
        dragAim.worldCurrentY = cy + camera.y;

        const p1 = players.find(p => p.id === 'p1');
        if (!p1 || p1.isDead) return;

        let angle = 0;
        let dist = 0;

        if (dragAimMode === 'slingshot') {
            const dx = p1.x - dragAim.worldCurrentX;
            const dy = p1.y - dragAim.worldCurrentY;
            angle = Math.atan2(dy, dx);
            dist = Math.hypot(dx, dy);
        } else {
            const dx = dragAim.worldCurrentX - p1.x;
            const dy = dragAim.worldCurrentY - p1.y;
            angle = Math.atan2(dy, dx);
            dist = Math.hypot(dx, dy);
        }

        dragAim.aimAngle = angle;
        dragAim.aimDist = dist;
        p1.setAim(angle);
    }

    function setupEventListeners() {
        window.addEventListener('keydown', (e) => {
            keys[e.code] = true;
            if (e.code === 'Space') {
                e.preventDefault();
                const p1 = players.find(p => p.id === 'p1');
                if (p1 && !p1.isDead) p1.fire();
            }
        });

        window.addEventListener('keyup', (e) => {
            keys[e.code] = false;
        });

        canvas.addEventListener('mousedown', (e) => {
            if (e.button !== 0) return;
            sound.init();
            isMouseDown = true;

            dragAim.active = true;
            const rect = canvas.getBoundingClientRect();
            dragAim.startX = e.clientX - rect.left;
            dragAim.startY = e.clientY - rect.top;
            updateDragAimCalculations(e.clientX, e.clientY);

            const p1 = players.find(p => p.id === 'p1');
            if (p1 && !p1.isDead && WEAPONS[p1.weaponType].isRapidFire) {
                p1.fire();
            }
        });

        window.addEventListener('mousemove', (e) => {
            const rect = canvas.getBoundingClientRect();
            mouse.x = e.clientX - rect.left;
            mouse.y = e.clientY - rect.top;
            mouse.worldX = mouse.x + camera.x;
            mouse.worldY = mouse.y + camera.y;

            if (dragAim.active) {
                updateDragAimCalculations(e.clientX, e.clientY);
            } else {
                const p1 = players.find(p => p.id === 'p1');
                if (p1 && !p1.isDead) {
                    p1.setAim(Math.atan2(mouse.worldY - p1.y, mouse.worldX - p1.x));
                }
            }
        });

        window.addEventListener('mouseup', (e) => {
            if (e.button !== 0) return;
            isMouseDown = false;
            if (dragAim.active) {
                const p1 = players.find(p => p.id === 'p1');
                if (p1 && !p1.isDead) {
                    p1.fire();
                }
                dragAim.active = false;
            }
        });

        // Touch input for aim/shoot
        canvas.addEventListener('touchstart', (e) => {
            sound.init();
            isMouseDown = true;
            for (let i = 0; i < e.changedTouches.length; i++) {
                const t = e.changedTouches[i];
                const rect = canvas.getBoundingClientRect();
                const tx = t.clientX - rect.left;

                if (tx < 180 && t.clientY > rect.height - 200) {
                    continue;
                }

                if (!dragAim.active) {
                    dragAim.active = true;
                    dragAim.identifier = t.identifier;
                    dragAim.startX = tx;
                    dragAim.startY = t.clientY - rect.top;
                    updateDragAimCalculations(t.clientX, t.clientY);

                    const p1 = players.find(p => p.id === 'p1');
                    if (p1 && !p1.isDead && WEAPONS[p1.weaponType].isRapidFire) {
                        p1.fire();
                    }
                    break;
                }
            }
        }, { passive: false });

        canvas.addEventListener('touchmove', (e) => {
            if (!dragAim.active) return;
            for (let i = 0; i < e.changedTouches.length; i++) {
                const t = e.changedTouches[i];
                if (t.identifier === dragAim.identifier) {
                    e.preventDefault();
                    updateDragAimCalculations(t.clientX, t.clientY);
                    break;
                }
            }
        }, { passive: false });

        const onTouchEnd = (e) => {
            if (!dragAim.active) return;
            isMouseDown = false;
            for (let i = 0; i < e.changedTouches.length; i++) {
                const t = e.changedTouches[i];
                if (t.identifier === dragAim.identifier) {
                    const p1 = players.find(p => p.id === 'p1');
                    if (p1 && !p1.isDead) p1.fire();
                    dragAim.active = false;
                    dragAim.identifier = null;
                    break;
                }
            }
        };

        canvas.addEventListener('touchend', onTouchEnd);
        canvas.addEventListener('touchcancel', onTouchEnd);

        const stickKnob = document.getElementById('touch-stick-knob');

        let stickCenter = { x: 0, y: 0 };

        function updateStick(clientX, clientY) {
            const maxRadius = 45;
            const dx = clientX - stickCenter.x;
            const dy = clientY - stickCenter.y;
            const dist = Math.hypot(dx, dy);
            const angle = Math.atan2(dy, dx);
            const clamped = Math.min(dist, maxRadius);

            const sx = Math.cos(angle) * clamped;
            const sy = Math.sin(angle) * clamped;

            stickKnob.style.transform = `translate(${sx}px, ${sy}px)`;
            touchMove.x = sx / maxRadius;
            touchMove.y = sy / maxRadius;
            touchMove.active = clamped > 5;
        }

        function endStick() {
            stickActiveId = null;
            touchMove.active = false;
            touchMove.x = 0;
            touchMove.y = 0;
            stickKnob.style.transform = `translate(0px, 0px)`;
        }

        stickZone.addEventListener('pointerdown', (e) => {
            e.preventDefault();
            sound.init();
            stickActiveId = e.pointerId;
            stickZone.setPointerCapture(e.pointerId);

            const rect = stickZone.getBoundingClientRect();
            stickCenter = {
                x: rect.left + rect.width / 2,
                y: rect.top + rect.height / 2
            };
            updateStick(e.clientX, e.clientY);
        });

        stickZone.addEventListener('pointermove', (e) => {
            if (e.pointerId !== stickActiveId) return;
            e.preventDefault();
            updateStick(e.clientX, e.clientY);
        });

        stickZone.addEventListener('pointerup', (e) => {
            if (e.pointerId !== stickActiveId) return;
            e.preventDefault();
            endStick();
        });

        stickZone.addEventListener('pointercancel', (e) => {
            if (e.pointerId !== stickActiveId) return;
            e.preventDefault();
            endStick();
        });

        const btnShoot = document.getElementById('btn-touch-shoot');
        btnShoot.addEventListener('pointerdown', (e) => {
            e.preventDefault();
            sound.init();
            isTouchFiring = true;
            const p1 = players.find(p => p.id === 'p1');
            if (p1 && !p1.isDead) p1.fire();
        });
        const stopTouchFire = (e) => {
            e.preventDefault();
            isTouchFiring = false;
        };
        btnShoot.addEventListener('pointerup', stopTouchFire);
        btnShoot.addEventListener('pointercancel', stopTouchFire);
        btnShoot.addEventListener('pointerleave', stopTouchFire);

        const btnBoost = document.getElementById('btn-touch-boost');
        btnBoost.addEventListener('pointerdown', (e) => {
            e.preventDefault();
            const p1 = players.find(p => p.id === 'p1');

            const p1 = players.find(p => p.id === 'p1');

        };

    }

    let lastTime = 0;
    function gameLoop(timestamp) {
        const dt = (timestamp - lastTime) / 1000 || 0.016;
        lastTime = timestamp;

        if (isRunning && !isPaused) {
            updateSimulation(dt);
        }

        renderSimulation();
        requestAnimationFrame(gameLoop);
    }

    function updateSimulation(dt) {
        const p1 = players.find(p => p.id === 'p1');

        if (p1 && !p1.isDead) {
            if (keys['KeyW'] || keys['ArrowUp']) p1.drive(true);
            if (keys['KeyS'] || keys['ArrowDown']) p1.drive(false);
            if (keys['KeyA'] || keys['ArrowLeft']) p1.turn(-1);
            if (keys['KeyD'] || keys['ArrowRight']) p1.turn(1);

            p1.isBoosting = keys['ShiftLeft'] || keys['ShiftRight'] || false;

            // On-screen joystick control input
            if (touchMove.active) {
                const targetHeading = Math.atan2(touchMove.y, touchMove.x);
                let diff = targetHeading - p1.heading;
                while (diff < -Math.PI) diff += Math.PI * 2;
                while (diff > Math.PI) diff -= Math.PI * 2;

                if (diff > 0.08) p1.turn(1);
                else if (diff < -0.08) p1.turn(-1);
                
                const mag = Math.hypot(touchMove.x, touchMove.y);
                if (mag > 0.15) {
                    p1.drive(true);
                }
            }
        }

        // Periodic multiplayer broadcast (20 times per second)
        const now = performance.now();
        if (now - lastNetworkSync > 50) {
            lastNetworkSync = now;
            broadcastLocalState();
        }

        // Bots
        for (let p of players) {
            if (p.isBot) updateShopperBotAI(p, dt);
            p.update(dt);
        }

        // Bullets
        for (let i = bullets.length - 1; i >= 0; i--) {
            const b = bullets[i];
            b.update();
            if (!b.isAlive) bullets.splice(i, 1);
        }

        // Fire puddles life
        for (let i = firePuddles.length - 1; i >= 0; i--) {
            const f = firePuddles[i];
            f.life--;
            if (f.life <= 0) firePuddles.splice(i, 1);
        }

        for (let i = proximityMines.length - 1; i >= 0; i--) {
            const m = proximityMines[i];
            m.update();
            if (m.life <= 0) proximityMines.splice(i, 1);
        }

        for (let i = shellCasings.length - 1; i >= 0; i--) {
            const sc = shellCasings[i];
            sc.update();
            if (sc.life <= 0) shellCasings.splice(i, 1);
        }

        for (let i = particles.length - 1; i >= 0; i--) {
            const part = particles[i];
            part.update();
            if (part.life <= 0) particles.splice(i, 1);
        }

        // Power-up Pickups
        for (let i = powerups.length - 1; i >= 0; i--) {
            const pu = powerups[i];
            for (let p of players) {
                if (p.isDead) continue;
                if (Math.hypot(p.x - pu.x, p.y - pu.y) < p.radius + pu.radius) {
                    p.equip(pu);
                    powerups.splice(i, 1);
                    break;
                }
            }
        }

        powerupSpawnCooldown -= dt;
        if (powerupSpawnCooldown <= 0) {
            powerupSpawnCooldown = 10 + Math.random() * 4;
            if (powerups.length < MAX_ACTIVE_POWERUPS) {
                spawnPowerup();
            }
        }

        // Camera Lerp
        if (p1) {
            const targetCamX = p1.x - camera.width / 2;
            const targetCamY = p1.y - camera.height / 2;
            camera.x += (targetCamX - camera.x) * 0.1;
            camera.y += (targetCamY - camera.y) * 0.1;

            camera.x = Math.max(0, Math.min(WORLD_WIDTH - camera.width, camera.x));
            camera.y = Math.max(0, Math.min(WORLD_HEIGHT - camera.height, camera.y));
        }

        updateHUD();
    }

    function renderSimulation() {
        ctx.clearRect(0, 0, camera.width, camera.height);

        ctx.save();
        ctx.translate(-camera.x, -camera.y);

        drawFloor(ctx);

        for (let track of tireTracks) {
            ctx.save();
            ctx.translate(track.x, track.y);
            ctx.rotate(track.angle);
            ctx.fillStyle = `rgba(15, 23, 42, ${track.alpha})`;
            ctx.fillRect(-3, -2, 6, 4);
            ctx.restore();
            track.life--;
            track.alpha *= 0.995;
        }
        tireTracks = tireTracks.filter(t => t.life > 0 && t.alpha > 0.02);

        for (let sc of shellCasings) sc.draw(ctx);
        for (let m of proximityMines) m.draw(ctx);

        drawObstacles(ctx);
        drawPowerups(ctx);

        for (let p of players) drawCart(ctx, p);

        drawDragAimIndicator(ctx);

        for (let b of bullets) b.draw(ctx);
        for (let part of particles) part.draw(ctx);

        ctx.restore();

        renderMinimap();
    }
